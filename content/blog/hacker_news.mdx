## Scraping the Data
The first step was gathering data. I decided to scrape Hacker
News every 10 minutes, pulling down posts and storing them in an Indexed DB.
From there, I dumped the data into a SQLite database, which was then pushed to a
server using a cron job. # Let's Look At Hacker News

## Scraping Script

```python
import requests
from bs4 import BeautifulSoup
import sqlite3
from datetime import datetime

# Create or connect to SQLite database
conn = sqlite3.connect('hacker_news.db')
c = conn.cursor()

# Create table for storing the scraped data
c.execute('''
    CREATE TABLE IF NOT EXISTS news (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        rank INTEGER,
        title TEXT,
        link TEXT,
        site TEXT,
        points INTEGER,
        author TEXT,
        comments_count INTEGER,
        item_id INTEGER,
        timestamp DATETIME DEFAULT CURRENT_TIMESTAMP
    )
''')

def scrape_hacker_news(page=1):
    url = f'https://news.ycombinator.com/news?p={page}'
    response = requests.get(url)
    soup = BeautifulSoup(response.text, 'html.parser')

    # Find all rows of news
    rows = soup.find_all('tr', class_='athing')

    news_data = []

    for row in rows:
        # Extract the rank
        rank = int(row.find('span', class_='rank').text.replace('.', ''))

        # Extract the title and link from the current row
        title_tag = row.find('span', class_='titleline').find('a')
        title = title_tag.text
        link = title_tag['href']

        # Extract the site (if available)
        site_tag = row.find_next_sibling('tr').find('span', class_='sitestr')
        site = site_tag.text if site_tag else 'N/A'

        # Get the subtext row for points, author, and comments
        subtext_row = row.find_next_sibling('tr').find('td', class_='subtext')

        # Extract points
        points_tag = subtext_row.find('span', class_='score')
        points = int(points_tag.text.replace(' points', '')) if points_tag else 0

        # Extract author, with None check
        author_tag = subtext_row.find('a', class_='hnuser')
        author = author_tag.text if author_tag else 'N/A'

        # Extract the number of comments and item ID
        try:
            comments_tag = subtext_row.find_all('a')[-1]
            comments_text = comments_tag.text.split('\xa0')[0]  # Extract comments count from text like "64 comments"
            comments_count = int(comments_text) if comments_text.isdigit() else 0
            item_id = int(comments_tag['href'].split('=')[-1])
        except Exception as e:
            comments_count = 0
            item_id = 0
            comments_text = 'N/A'
            comments_tag = None

        news_data.append((rank, title, link, site, points, author, comments_count, item_id))


    print(f"Scraped {len(news_data)} news from page {page}")
    # Insert data into SQLite
    c.executemany('INSERT INTO news (rank, title, link, site, points, author, comments_count, item_id) VALUES (?, ?, ?, ?, ?, ?, ?, ?)', news_data)
    conn.commit()

    # Check if there's a 'more' link to continue pagination
    more_link = soup.find('a', class_='morelink')
    print("More link found:", more_link is not None)
    if more_link:
        next_page = page + 1
        scrape_hacker_news(next_page)  # Recursive call to scrape the next page

# Function to fetch the history of articles over time
def fetch_news_history():
    c.execute('SELECT * FROM news ORDER BY timestamp DESC')
    rows = c.fetchall()
    for row in rows:
        print(row)

# Run the scraper
if __name__ == '__main__':
    scrape_hacker_news()
    fetch_news_history()

# Close the database connection when done
conn.close()
```
